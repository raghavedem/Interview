request timing middleware
@app.middleware("http")
async def timing(request, call_next):
    t = time.perf_counter()
    response = await call_next(request)
    response.headers["X-Time"] = \
        f"{time.perf_counter()-t:.3f}s"
    return response

Postgres SQL patterns
window functions
SELECT user_id, amount,
  SUM(amount) OVER (
    PARTITION BY user_id
    ORDER BY created_at
  ) AS running_total,
  ROW_NUMBER() OVER (
    PARTITION BY user_id
    ORDER BY amount DESC
  ) AS rank_by_amount
FROM orders;
recursive CTE (hierarchies)
WITH RECURSIVE tree AS (
  SELECT id, name, parent_id, 0 AS depth
  FROM categories WHERE parent_id IS NULL
  UNION ALL
  SELECT c.id, c.name, c.parent_id,
         t.depth + 1
  FROM categories c
  JOIN tree t ON c.parent_id = t.id
)
SELECT * FROM tree;
upsert
INSERT INTO stats(uid, cnt) VALUES ($1, 1)
ON CONFLICT (uid) DO UPDATE
  SET cnt = stats.cnt + 1
RETURNING *;
LATERAL join (top-N per group)
SELECT u.id, recent.order_id
FROM users u
CROSS JOIN LATERAL (
  SELECT id AS order_id FROM orders
  WHERE user_id = u.id
  ORDER BY created_at DESC
  LIMIT 3
) recent;
